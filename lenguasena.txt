#include <Wire.h>
#include <MPU6050.h>
#include <math.h>

MPU6050 mpu;

// ─────────────────────────────────────
// Pines Flex
// ─────────────────────────────────────
#define FLEX_PULGAR  A0
#define FLEX_INDICE  A1
#define FLEX_MEDIO   A2
#define FLEX_ANULAR  A3
#define FLEX_MENIQUE A4

#define N_MUESTRAS 5

// ─────────────────────────────────────
// FILTROS
// ─────────────────────────────────────
float alphaFlex = 0.7;
float alphaMPU  = 0.85;

float flexFiltrado[5] = {0,0,0,0,0};

float pitchFiltrado = 0;
float rollFiltrado  = 0;

// ─────────────────────────────────────
// Calibración
// ─────────────────────────────────────
int calibAbierto[5]  = {831, 803, 796, 782, 629};
int calibCerrado[5]  = {927, 894, 915, 898, 960};
int thresholdFlex[5] = {879, 848, 855, 840, 794};

int margen = 6;

// ─────────────────────────────────────
// MPU Base
// ─────────────────────────────────────
float basePitch = 0;
float baseRoll  = 0;

// ─────────────────────────────────────
// Estado
// ─────────────────────────────────────
int dedos[5] = {0,0,0,0,0};

String ultimaLetra = "";
String ultimaLetraImpresa = "";

int contadorEstable = 0;
int contadorRepeticion = 0;

const int ESTABILIDAD_REQUERIDA = 3;
const int REPETICION_LETRA = 20;

// ─────────────────────────────────────
// Leer promedio flex
// ─────────────────────────────────────
int leerPromedio(int pin){

  long suma = 0;

  for(int i = 0; i < N_MUESTRAS; i++){
    suma += analogRead(pin);
    delay(3);
  }

  return suma / N_MUESTRAS;
}

// ─────────────────────────────────────
// Calcular Pitch y Roll
// ─────────────────────────────────────
void calcAngulos(
  int16_t ax,
  int16_t ay,
  int16_t az,
  float &pitch,
  float &roll
){

  float axf = ax / 16384.0;
  float ayf = ay / 16384.0;
  float azf = az / 16384.0;

  pitch = atan2(
            ayf,
            sqrt(axf * axf + azf * azf)
          ) * 180.0 / PI;

  roll = atan2(
           -axf,
           azf
         ) * 180.0 / PI;
}

// ─────────────────────────────────────
// Clasificación 0 y 1
// ─────────────────────────────────────
void clasificarDedos(int flex[]){

  for(int i = 0; i < 5; i++){

    int margenActual = margen;

    if(i == 4){
      margenActual = 0;
    }

    if(calibCerrado[i] > calibAbierto[i]){

      if(flex[i] > thresholdFlex[i] + margenActual){
        dedos[i] = 1;
      }
      else if(flex[i] < thresholdFlex[i] - margenActual){
        dedos[i] = 0;
      }

    } else {

      if(flex[i] < thresholdFlex[i] - margenActual){
        dedos[i] = 1;
      }
      else if(flex[i] > thresholdFlex[i] + margenActual){
        dedos[i] = 0;
      }
    }
  }
}

// ─────────────────────────────────────
// Estado neutral
// ─────────────────────────────────────
bool esNeutral(int flex[]){

  for(int i = 0; i < 5; i++){

    if(abs(flex[i] - calibAbierto[i]) > 25){
      return false;
    }
  }

  return true;
}

// ─────────────────────────────────────
// LETRA A
// ─────────────────────────────────────
String detectarA(){

  if(
    dedos[0] == 0 &&
    dedos[1] == 1 &&
    dedos[2] == 1 &&
    dedos[3] == 1 &&
    dedos[4] == 1
  ){
    return "A";
  }

  return "-";
}

}

// ─────────────────────────────────────
// LETRA C
// ─────────────────────────────────────
String detectarC(
  int flex[],
  float pitch,
  float roll
){

  if(

    flex[0] > 900 &&
    flex[1] > 885 &&
    flex[2] > 900 &&

    flex[3] > 840 &&
    flex[3] < 870 &&

    flex[4] > 900

  ){

    float dRoll = roll - baseRoll;

    if(dRoll < 105){
      return "C";
    }
  }

  return "-";
}

// ─────────────────────────────────────
// LETRA D
// ─────────────────────────────────────
String detectarD(
  int flex[],
  float pitch,
  float roll
){

  if(
    dedos[0] != 1 ||
    dedos[1] != 0 ||
    dedos[2] != 1 ||
    dedos[3] != 1 ||
    dedos[4] != 1
  ){
    return "-";
  }

  float dPitch = pitch - basePitch;
  float dRoll  = roll  - baseRoll;

  if(
    abs(dPitch) < 40 &&
    abs(dRoll)  < 125
  ){
    return "D";
  }

  return "-";
}

// ─────────────────────────────────────
// DETECCIÓN GENERAL
// ─────────────────────────────────────
String detectarLetra(
  int flex[],
  float pitch,
  float roll
){

  if(esNeutral(flex)){
    return "-";
  }

  String c = detectarC(flex, pitch, roll);
  if(c == "C") return "C";

  String a = detectarA();
  if(a == "A") return "A";

  String b = detectarB();
  if(b == "B") return "B";

  String d = detectarD(flex, pitch, roll);
  if(d == "D") return "D";

  String e = detectarE(roll);
  if(e == "E") return "E";

  String f = detectarF();
  if(f == "F") return "F";

  String g = detectarG(pitch, roll);
  if(g == "G") return "G";

  String h = detectarH(pitch, roll);
  if(h == "H") return "H";

  String i = detectarI();
  if(i == "I") return "I";

  String l = detectarL();
  if(l == "L") return "L";

  String m = detectarM(pitch, roll);
  if(m == "M") return "M";

  String n = detectarN(pitch, roll);
  if(n == "N") return "N";

  String o = detectarO(flex, pitch, roll);
  if(o == "O") return "O";

  String p = detectarP(pitch, roll);
  if(p == "P") return "P";

  String s = detectarS(flex);
  if(s == "S") return "S";

  String w = detectarW(pitch, roll);
  if(w == "W") return "W";

  String y = detectarY();
  if(y == "Y") return "Y";

  String u = detectarU();
  if(u == "U") return "U";

  return "-";
}

// ─────────────────────────────────────
// SETUP
// ─────────────────────────────────────
void setup(){

  Serial.begin(9600);

  Wire.begin();

  mpu.initialize();

  Serial.println("\nCalibrando MPU...");
  Serial.println("NO muevas la mano\n");

  delay(3000);

  long sumaPitch = 0;
  long sumaRoll  = 0;

  for(int i = 0; i < 100; i++){

    int16_t ax, ay, az;

    mpu.getAcceleration(&ax, &ay, &az);

    float pitch, roll;

    calcAngulos(ax, ay, az, pitch, roll);

    sumaPitch += pitch;
    sumaRoll  += roll;

    delay(20);
  }

  basePitch = sumaPitch / 100.0;
  baseRoll  = sumaRoll  / 100.0;

  Serial.print("Base Pitch: ");
  Serial.println(basePitch);

  Serial.print("Base Roll: ");
  Serial.println(baseRoll);

  Serial.println("\nSistema listo\n");
}

// ─────────────────────────────────────
// LOOP
// ─────────────────────────────────────
void loop(){

  int pins[5] = {
    FLEX_PULGAR,
    FLEX_INDICE,
    FLEX_MEDIO,
    FLEX_ANULAR,
    FLEX_MENIQUE
  };

  int flex[5];

  // ─────────────────
  // Leer FLEX + FILTRO
  // ─────────────────
  for(int i = 0; i < 5; i++){

    int lectura = leerPromedio(pins[i]);

    flexFiltrado[i] =
      alphaFlex * flexFiltrado[i] +
      (1.0 - alphaFlex) * lectura;

    flex[i] = (int)flexFiltrado[i];
  }

  clasificarDedos(flex);

  // ─────────────────
  // MPU
  // ─────────────────
  int16_t ax, ay, az;

  mpu.getAcceleration(&ax, &ay, &az);

  float pitch, roll;

  calcAngulos(ax, ay, az, pitch, roll);

  // ─────────────────
  // FILTRO MPU
  // ─────────────────
  pitchFiltrado =
    alphaMPU * pitchFiltrado +
    (1.0 - alphaMPU) * pitch;

  rollFiltrado =
    alphaMPU * rollFiltrado +
    (1.0 - alphaMPU) * roll;

  // ─────────────────
  // Detectar letra
  // ─────────────────
  String letra = detectarLetra(
                   flex,
                   pitchFiltrado,
                   rollFiltrado
                 );

  // ─────────────────
  // Estabilidad
  // ─────────────────
  if(letra == ultimaLetra){
    contadorEstable++;
  }
  else{
    contadorEstable = 0;
  }

  // ─────────────────
  // Mostrar UNA sola vez
  // ─────────────────
  if(
    contadorEstable >= ESTABILIDAD_REQUERIDA &&
    letra != "-"
  ){

    if(letra != ultimaLetraImpresa){

      Serial.print(">>> LETRA: ");
      Serial.println(letra);

      ultimaLetraImpresa = letra;

      contadorRepeticion = 0;
    }
    else{

      contadorRepeticion++;

      if(contadorRepeticion >= REPETICION_LETRA){

        Serial.print(">>> LETRA: ");
        Serial.println(letra);

        contadorRepeticion = 0;
      }
    }

    contadorEstable = 0;
  }

  // ─────────────────
  // Reset si mano neutra
  // ─────────────────
  if(letra == "-"){
    ultimaLetraImpresa = "";
  }

  ultimaLetra = letra;

  delay(120);
}
